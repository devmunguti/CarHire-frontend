        {filtered.map((entry) => expanded[entry.id] && (
          <div className="audit-diff" key={`${entry.id}-diff`}>
            <h3>Before</h3><pre>{JSON.stringify(entry.before ?? {}, null, 2)}</pre>
            <h3>After</h3><pre>{JSON.stringify(entry.after ?? {}, null, 2)}</pre>
          </div>
        ))}
      </div>
    </section>
  )
}
